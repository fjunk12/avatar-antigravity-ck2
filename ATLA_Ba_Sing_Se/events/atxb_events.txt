# ATLA Ba Sing Se - events (namespace atxb)
# atxb.1   yearly run, once a year for the whole world (on_yearly_pulse, hidden)
# atxb.3   a foreign or Way of Chin ruler loots the treasury of Ba Sing Se
# atxb.5   "The Treasury of Ba Sing Se" overview (decision)
# atxb.6   the Grand Secretariat's report to a puppet Earth King (players)
# atxb.10  a work completes a stage            atxb.11  a work stalls or is abandoned
# atxb.15  a work is finished                   atxb.16  news to tributaries: A Single Law for the Kingdom
# atxb.18  sabotage: what came of it (saboteur) atxb.19  sabotage: the agents were caught (king, FROM = saboteur)
# atxb.20  a tributary breaks away (hidden, ROOT = tributary)
# atxb.28  a player's plea (hidden, ROOT = pleader, from the decision)
# atxb.29  an AI plea (hidden, ROOT = pleader; the yearly cap is counted here)
# atxb.31  Plea to Ba Sing Se: the king answers (ROOT = king, FROM = pleader)
# atxb.32  the pleader hears the answer (letter)
# atxb.33  memorandum to a puppet king: the Grand Secretariat answered a plea
namespace = atxb

character_event = {
	id = atxb.1
	hide_window = yes
	is_triggered_only = yes # on_yearly_pulse

	only_rulers = yes

	trigger = {
		OR = {
			NOT = { has_global_flag = atxb_tick }
			had_global_flag = { flag = atxb_tick days = 330 }
		}
	}

	immediate = {
		clr_global_flag = atxb_tick
		set_global_flag = atxb_tick # one character runs this each year
		if = {
			limit = { NOT = { has_global_flag = atxb_bss_started } }
			set_global_flag = atxb_bss_started
		}
		set_variable = { which = global_atxb_breaks value = 0 }
		set_variable = { which = global_atxb_pleas value = 0 }

		# A foreign or Way of Chin ruler holding the Royal Palace loots the treasury
		if = {
			limit = {
				check_variable = { which = global_atxb_treasury value = 1 }
				c_royal_earth_palace = {
					has_holder = yes
					holder_scope = { top_liege = { atxb_valid_king_trigger = no } }
				}
			}
			c_royal_earth_palace = {
				holder_scope = {
					top_liege = {
						atxb_loot_treasury_effect = yes
						log = "ATLA Ba Sing Se: [This.GetTitledName] looted the treasury"
						character_event = { id = atxb.3 }
					}
				}
			}
		}

		atxb_find_king_effect = yes
		if = {
			limit = { event_target:atxb_king = { always = yes } }
			atxb_law_watch_effect = yes
			event_target:atxb_king = {
				atxb_yearly_income_effect = yes
				atxb_progress_all_works_effect = yes
				if = {
					limit = { trait = earthkingpuppet }
					atxb_secretariat_effect = yes
				}
				# The reach of Ba Sing Se: which tributaries are connected to the capital
				atxb_flood_fill_effect = yes
				log = "ATLA Ba Sing Se: [This.GetTitledName] crown [global_atxb_last_crown.GetValue], skim [global_atxb_last_skim.GetValue] from [global_atxb_tribs.GetValue] tributaries ([global_atxb_unlinked.GetValue] out of reach), treasury [global_atxb_treasury.GetValue], Royal Authority [global_royalAuthority.GetValue]"

				# Breakaways (at most 2 a year): twice as likely after the crown refused their plea, half as likely after A Single Law
				any_tributary = {
					limit = { atxb_breakaway_candidate_trigger = yes }
					if = {
						limit = {
							has_character_flag = atxb_refused_plea
							NOT = { had_character_flag = { flag = atxb_refused_plea days = 3650 } }
						}
						if = {
							limit = { has_global_flag = atxb_done_law }
							random = { chance = 20 character_event = { id = atxb.20 } }
						}
						else = {
							random = { chance = 40 character_event = { id = atxb.20 } }
						}
					}
					else_if = {
						limit = { has_global_flag = atxb_done_law }
						random = { chance = 10 character_event = { id = atxb.20 } }
					}
					else = {
						random = { chance = 20 character_event = { id = atxb.20 } }
					}
				}
				atxb_unlink_all_effect = yes
			}

			# Plea to Ba Sing Se: AI Earth realms under attack ask the crown for help (at most 2 a year, counted in atxb.29)
			any_independent_ruler = {
				limit = {
					ai = yes
					war = yes
					atxb_plea_base_trigger = yes
				}
				random = { chance = 25 character_event = { id = atxb.29 } }
			}
		}
	}
}

	id = atxb.18
	desc = { trigger = { has_character_flag = atxb_sab_hit } text = atxb.18.d_hit }
	desc = { trigger = { has_character_flag = atxb_sab_nothing } text = atxb.18.d_nothing }
	desc = { trigger = { has_character_flag = atxb_sab_caught } text = atxb.18.d_caught }
	picture = GFX_evt_ba_sing_se
	is_triggered_only = yes
	hide_window = no

	immediate = {
		atxb_find_king_effect = yes
		random_list = {
			45 = {
				modifier = { factor = 0 atxb_any_work_running_trigger = no }
				set_character_flag = atxb_sab_hit
				atxb_sabotage_setback_effect = yes
			}
			30 = {
				set_character_flag = atxb_sab_nothing
			}
			25 = {
				modifier = { factor = 0.5 intrigue = 15 }
				modifier = { factor = 1.5 event_target:atxb_king = { job_spymaster = { intrigue = 15 } } }
				set_character_flag = atxb_sab_caught
				prestige = -50
				event_target:atxb_king = {
					opinion = { who = ROOT modifier = opinion_atxb_saboteur years = 10 }
					character_event = { id = atxb.19 days = 2 }
				}
			}
		}
		log = "ATLA Ba Sing Se: [Root.GetTitledName] sabotaged the crown's works"
	}

	option = {
		name = atxb.18.a
	}
	after = {
		clr_character_flag = atxb_sab_hit
		clr_character_flag = atxb_sab_nothing
		clr_character_flag = atxb_sab_caught
		clr_character_flag = atxb_sab_roads
		clr_character_flag = atxb_sab_law
		clr_character_flag = atxb_sab_granaries

# Plea to Ba Sing Se: the king answers (ROOT = king, FROM = pleader). Aid is paid from the treasury, or from the king's
# own purse when the treasury cannot cover it. How much the crown can send depends on Royal Authority:
# below 30: 25 gold / 300 men; 30-59: 40 gold / 600 men; 60+: 60 gold / 1,000 men. The host is raised at the
# pleader's capital, costs no upkeep and disbands at peace.
character_event = {
	id = atxb.31
	desc = atxb.31.desc
	picture = GFX_evt_ba_sing_se
	is_triggered_only = yes

	immediate = {
		atxb_find_king_effect = yes
	}

	option = { # Send gold
		name = atxb.31.a
		trigger = {
			OR = {
				AND = {
					atxb_band_weak_trigger = yes
					atxb_crown_can_pay_25_trigger = yes
				}
				AND = {
					atxb_band_steady_trigger = yes
					atxb_crown_can_pay_40_trigger = yes
				}
				AND = {
					atxb_band_strong_trigger = yes
					atxb_crown_can_pay_60_trigger = yes
				}
			}
		}
		if = {
			limit = { atxb_band_weak_trigger = yes }
			atxb_crown_pays_25_effect = yes
			FROM = { wealth = 25 }
		}
		else_if = {
			limit = { atxb_band_steady_trigger = yes }
			atxb_crown_pays_40_effect = yes
			FROM = { wealth = 40 }
		}
		else_if = {
			limit = { atxb_band_strong_trigger = yes }
			atxb_crown_pays_60_effect = yes
			FROM = { wealth = 60 }
		}
		change_variable = { which = global_royalAuthority value = 1 }
		atxb_clamp_authority_effect = yes
		FROM = {
			atxb_plea_helped_effect = yes
			opinion = { who = ROOT modifier = opinion_atxb_helped years = 10 }
			set_character_flag = atxb_ans_gold
			letter_event = { id = atxb.32 days = 10 }
		}
		ai_chance = {
			factor = 30
			modifier = { factor = 0.5 trait = greedy }
		}
	}
	option = { # Raise a host at their capital
		name = atxb.31.b
		trigger = {
			OR = {
				AND = {
					atxb_band_weak_trigger = yes
					atxb_crown_can_pay_30_trigger = yes
				}
				AND = {
					atxb_band_steady_trigger = yes
					atxb_crown_can_pay_50_trigger = yes
				}
				AND = {
					atxb_band_strong_trigger = yes
					atxb_crown_can_pay_75_trigger = yes
				}
			}
		}
		if = {
			limit = { atxb_band_weak_trigger = yes }
			atxb_crown_pays_30_effect = yes
			FROM = { atxb_raise_host_small_effect = yes }
		}
		else_if = {
			limit = { atxb_band_steady_trigger = yes }
			atxb_crown_pays_50_effect = yes
			FROM = { atxb_raise_host_medium_effect = yes }
		}
		else_if = {
			limit = { atxb_band_strong_trigger = yes }
			atxb_crown_pays_75_effect = yes
			FROM = { atxb_raise_host_large_effect = yes }
		}
		change_variable = { which = global_royalAuthority value = 1 }
		atxb_clamp_authority_effect = yes
		FROM = {
			atxb_plea_helped_effect = yes
			opinion = { who = ROOT modifier = opinion_atxb_helped years = 10 }
			set_character_flag = atxb_ans_host
			letter_event = { id = atxb.32 days = 10 }
		}
		ai_chance = {
			factor = 40
			modifier = { factor = 1.5 trait = brave }
		}
	}
	option = { # Gold and a host
		name = atxb.31.c
		trigger = {
			OR = {
				AND = {
					atxb_band_weak_trigger = yes
					atxb_crown_can_pay_55_trigger = yes
				}
				AND = {
					atxb_band_steady_trigger = yes
					atxb_crown_can_pay_90_trigger = yes
				}
				AND = {
					atxb_band_strong_trigger = yes
					atxb_crown_can_pay_135_trigger = yes
				}
			}
		}
		if = {
			limit = { atxb_band_weak_trigger = yes }
			atxb_crown_pays_55_effect = yes
			FROM = {
				wealth = 25
				atxb_raise_host_small_effect = yes
			}
		}
		else_if = {
			limit = { atxb_band_steady_trigger = yes }
			atxb_crown_pays_90_effect = yes
			FROM = {
				wealth = 40
				atxb_raise_host_medium_effect = yes
			}
		}
		else_if = {
			limit = { atxb_band_strong_trigger = yes }
			atxb_crown_pays_135_effect = yes
			FROM = {
				wealth = 60
				atxb_raise_host_large_effect = yes
			}
		}
		change_variable = { which = global_royalAuthority value = 2 }
		atxb_clamp_authority_effect = yes
		FROM = {
			atxb_plea_helped_effect = yes
			opinion = { who = ROOT modifier = opinion_atxb_helped_both years = 10 }
			set_character_flag = atxb_ans_both
			letter_event = { id = atxb.32 days = 10 }
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 2 check_variable = { which = global_atxb_treasury value = 300 } }
			modifier = { factor = 1.5 trait = kind }
		}
	}
	option = { # Refuse
		name = atxb.31.d
		FROM = {
			atxb_plea_refused_effect = yes
			set_character_flag = atxb_ans_refused
			letter_event = { id = atxb.32 days = 10 }
		}
		ai_chance = {
			factor = 10
			modifier = { factor = 3 trait = greedy }
			modifier = { factor = 2 trait = craven }
			modifier = { factor = 0.5 trait = kind }
		}
	}
}
