# ATLA Ba Sing Se - decisions (prefix atxb_)
# Players open the Treasury of Ba Sing Se (one high-priority entry in the Intrigue tab); the AI takes the works, the draw and the corvee as decisions.
# A puppet Earth King (trait earthkingpuppet) cannot spend the treasury: the Grand Secretariat does (see atxb_secretariat_effect).
decisions = {
	# Players: one entry in the Intrigue tab opens the treasury popup (atxb.5), where the works, Draw on the Treasury and the corvee live.
	atxb_view_treasury = {
		ai = no
		is_high_prio = yes
		potential = { atxb_is_king_trigger = yes }
		allow = { always = yes }
		effect = { character_event = { id = atxb.5 } }
		ai_will_do = { factor = 0 }
	}

	# The AI uses these decisions; players use the popup
	atxb_work_roads = {
		ai_check_interval = 12
		potential = {
			ai = yes
			atxb_can_start_roads_trigger = yes
			atxb_any_work_running_trigger = no # the AI builds one work at a time
			check_variable = { which = global_atxb_treasury value = 225 } # first payment plus one yearly stage in hand
		}
		allow = { always = yes }
		effect = {
			atxb_start_roads_effect = yes
			custom_tooltip = { text = atxb_work_roads_tt }
		}
		ai_will_do = { factor = 1 }
	}
	atxb_work_law = {
		ai_check_interval = 12
		potential = {
			ai = yes
			atxb_can_start_law_trigger = yes
			atxb_any_work_running_trigger = no # the AI builds one work at a time
			check_variable = { which = global_atxb_treasury value = 400 } # first payment plus one yearly stage in hand
		}
		allow = { always = yes }
		effect = {
			atxb_start_law_effect = yes
			custom_tooltip = { text = atxb_work_law_tt }
		}
		ai_will_do = { factor = 1 }
	}
	atxb_work_granaries = {
		ai_check_interval = 12
		potential = {
			ai = yes
			atxb_can_start_granaries_trigger = yes
			atxb_any_work_running_trigger = no # the AI builds one work at a time
			check_variable = { which = global_atxb_treasury value = 200 } # first payment plus one yearly stage in hand
		}
		allow = { always = yes }
		effect = {
			atxb_start_granaries_effect = yes
			custom_tooltip = { text = atxb_work_granaries_tt }
		}
		ai_will_do = { factor = 1 }
	}
	atxb_draw_on_treasury = {
		ai_check_interval = 24
		potential = {
			ai = yes
			atxb_can_draw_trigger = yes
		}
		allow = { always = yes }
		effect = { atxb_draw_effect = yes }
		ai_will_do = {
			factor = 0.2
			modifier = { factor = 5 trait = greedy }
			modifier = { factor = 0 trait = charitable }
			modifier = { factor = 3 NOT = { wealth = 0 } }
		}
	}
	atxb_corvee = {
		ai_check_interval = 12
		potential = {
			ai = yes
			atxb_can_corvee_trigger = yes
		}
		allow = { always = yes }
		effect = { atxb_corvee_effect = yes }
		ai_will_do = {
			factor = 0.5
			modifier = { factor = 0 NOT = { check_variable = { which = global_royalAuthority value = 50 } } } # keeps a reserve
			modifier = { factor = 2 check_variable = { which = global_royalAuthority value = 90 } }
		}
	}

	# Rival Earth realms and Way of Chin realms of the Earth Kingdom can pay to disrupt the crown's works
	atxb_sabotage_works = {
		ai_check_interval = 60
		potential = {
			is_landed = yes
			is_ruler = yes
			independent = yes
			higher_tier_than = COUNT
			atxb_is_king_trigger = no
			atxb_any_work_running_trigger = yes
			NOR = {
				mercenary = yes
				holy_order = yes
				rebel = yes
			}
			OR = {
				culture_group = earth_kingdom_cultures
				religion = way_of_chin
			}
			capital_scope = { region = world_greater_earth_kingdom }
		}
		allow = {
			wealth = 25
			custom_tooltip = {
				text = atxb_sabotage_cooldown_tt
				OR = {
					NOT = { has_character_flag = atxb_sabotaged }
					had_character_flag = { flag = atxb_sabotaged days = 1825 }
				}
			}
		}
		effect = {
			wealth = -25
			clr_character_flag = atxb_sabotaged
			set_character_flag = atxb_sabotaged
			custom_tooltip = { text = atxb_sabotage_tt }
			character_event = { id = atxb.18 days = 30 }
		}
		ai_will_do = {
			factor = 0.05
			modifier = { factor = 0 atxb_hostile_to_crown_trigger = no } # only realms that dislike the king, are his rival, or follow the Way of Chin
			modifier = { factor = 4 religion = way_of_chin }
			modifier = { factor = 3 trait = ambitious }
			modifier = { factor = 2 trait = deceitful }
			modifier = { factor = 0.2 trait = honest }
			modifier = { factor = 0 NOT = { wealth = 100 } }
		}
	}

	# Plea to Ba Sing Se (players): an Earth realm under attack by the Fire Nation, the Way of Chin or rebels asks the crown for help.
	# The AI pleads through the yearly run. Player pleas do not count toward the yearly cap.
	atxb_plea_to_ba_sing_se = {
		ai = no
		potential = {
			war = yes
			atxb_plea_base_trigger = yes
		}
		allow = {
			atxb_plea_war_trigger = yes
		}
		effect = {
			custom_tooltip = { text = atxb_plea_tt }
			character_event = { id = atxb.28 }
		}
		ai_will_do = { factor = 0 }
	}

	# Hire Daofei Irregulars: an Earth lord outside Ba Sing Se at war buys the outlaws' swords (and lets them settle in)
	atxb_hire_daofei = {
		ai_check_interval = 12
		potential = {
			war = yes
			atxb_daofei_ruler_trigger = yes
		}
		allow = {
			wealth = 50
			custom_tooltip = {
				text = atxb_daofei_cooldown_tt
				OR = {
					NOT = { has_character_flag = atxb_hired_daofei }
					had_character_flag = { flag = atxb_hired_daofei days = 1095 }
				}
			}
		}
		effect = {
			wealth = -50
			clr_character_flag = atxb_hired_daofei
			set_character_flag = atxb_hired_daofei
			custom_tooltip = { text = atxb_hire_daofei_tt }
			hidden_tooltip = {
				atxb_daofei_irregulars_effect = yes
				if = {
					limit = { atxb_has_daofei_trigger = no }
					capital_scope = { save_event_target_as = atxb_daofei_prov }
					atxb_daofei_settle_effect = yes
				}
			}
			log = "ATLA Ba Sing Se: [Root.GetTitledName] hired daofei irregulars"
		}
		ai_will_do = {
			factor = 0.2
			modifier = { factor = 0 NOT = { wealth = 150 } }
			modifier = { factor = 0.3 trait = honest }
			modifier = { factor = 2 atxb_has_daofei_trigger = yes }
		}
	}

	# Petition for a Royal Grant (players): an Earth Kingdom tributary asks the crown for gold from the treasury of Ba Sing Se.
	# AI tributaries ask through the yearly run (at most 2 a year).
	atxb_request_royal_grant = {
		ai = no
		potential = {
			atxb_grant_base_trigger = yes
		}
		allow = {
			custom_tooltip = {
				text = atxb_grant_treasury_tt
				check_variable = { which = global_atxb_treasury value = 50 }
			}
		}
		effect = {
			custom_tooltip = { text = atxb_grant_tt }
			character_event = { id = atxb.50 }
		}
		ai_will_do = { factor = 0 }
	}
}

# Royal Arbitration: the Earth King orders an Earth realm to end its war against another Earth realm (diplomacy screen of the attacker)
targetted_decisions = {
	atxb_royal_arbitration = {
		ai_check_interval = 12
		filter = independent_rulers
		ai_target_filter = independent_rulers
		from_potential = {
			atxb_king_can_spend_trigger = yes
		}
		potential = {
			atxb_arbitration_target_trigger = yes
		}
		allow = {
			FROM = {
				custom_tooltip = {
					text = atxb_arb_cooldown_tt
					OR = {
						NOT = { has_character_flag = atxb_arbitrated }
						had_character_flag = { flag = atxb_arbitrated days = 365 }
					}
				}
			}
			conditional_tooltip = {
				trigger = { tier = DUKE }
				custom_tooltip = {
					text = atxb_arb_cost_10_tt
					hidden_tooltip = { check_variable = { which = global_royalAuthority value = 10 } }
				}
			}
			conditional_tooltip = {
				trigger = { tier = KING }
				custom_tooltip = {
					text = atxb_arb_cost_15_tt
					hidden_tooltip = { check_variable = { which = global_royalAuthority value = 15 } }
				}
			}
			conditional_tooltip = {
				trigger = { tier = EMPEROR }
				custom_tooltip = {
					text = atxb_arb_cost_25_tt
					hidden_tooltip = { check_variable = { which = global_royalAuthority value = 25 } }
				}
			}
		}
		effect = {
			FROM = {
				if = {
					limit = { ROOT = { tier = DUKE } }
					custom_tooltip = {
						text = atxb_arb_spend_10_tt
						hidden_tooltip = { change_variable = { which = global_royalAuthority value = -10 } }
					}
				}
				if = {
					limit = { ROOT = { tier = KING } }
					custom_tooltip = {
						text = atxb_arb_spend_15_tt
						hidden_tooltip = { change_variable = { which = global_royalAuthority value = -15 } }
					}
				}
				if = {
					limit = { ROOT = { tier = EMPEROR } }
					custom_tooltip = {
						text = atxb_arb_spend_25_tt
						hidden_tooltip = { change_variable = { which = global_royalAuthority value = -25 } }
					}
				}
				atxb_clamp_authority_effect = yes
				clr_character_flag = atxb_arbitrated
				set_character_flag = atxb_arbitrated
				save_global_event_target_as = atxb_arbiter
			}
			save_global_event_target_as = atxb_arb_target
			custom_tooltip = { text = atxb_arbitration_tt }
			log = "ATLA Ba Sing Se: [From.GetTitledName] demanded that [Root.GetTitledName] make peace"
			character_event = { id = atxb.42 }
		}
		# AI kings arbitrate rarely, only with a strong crown (Royal Authority 60+); FROM is the king
		ai_will_do = {
			factor = 0.05
			modifier = { factor = 0 NOT = { check_variable = { which = global_royalAuthority value = 60 } } }
			modifier = { factor = 2 check_variable = { which = global_royalAuthority value = 90 } }
			modifier = { factor = 2 pays_tribute_to = FROM }
		}
	}
}
