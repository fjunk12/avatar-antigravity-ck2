# ATLA Ba Sing Se - effects (prefix atxb_)
# Treasury: global_atxb_treasury (0-1000). Readouts: global_atxb_last_crown, global_atxb_last_skim, global_atxb_tribs.

# Saves the ruler of Ba Sing Se as event_target:atxb_king (if there is a valid one) and province 109 as event_target:atxb_bss
atxb_find_king_effect = {
	109 = { save_event_target_as = atxb_bss }
	if = {
		limit = { e_earth_kingdom = { has_holder = yes holder_scope = { atxb_valid_king_trigger = yes } } }
		e_earth_kingdom = { holder_scope = { save_event_target_as = atxb_king } }
	}
	else_if = {
		limit = {
			NOT = { e_earth_kingdom = { has_holder = yes } }
			k_earth_kingdom = { has_holder = yes holder_scope = { atxb_valid_king_trigger = yes } }
		}
		k_earth_kingdom = { holder_scope = { save_event_target_as = atxb_king } }
	}
}

atxb_clamp_treasury_effect = {
	if = {
		limit = { check_variable = { which = global_atxb_treasury value = 1000 } }
		set_variable = { which = global_atxb_treasury value = 1000 }
	}
	if = {
		limit = { NOT = { check_variable = { which = global_atxb_treasury value = 0 } } }
		set_variable = { which = global_atxb_treasury value = 0 }
	}
}

# Keeps Royal Authority in the base mod's range of -100 to 100 (earthking.56 uses the same bounds)
atxb_clamp_authority_effect = {
	if = {
		limit = { check_variable = { which = global_royalAuthority value = 100 } }
		set_variable = { which = global_royalAuthority value = 100 }
	}
	if = {
		limit = { NOT = { check_variable = { which = global_royalAuthority value = -100 } } }
		set_variable = { which = global_royalAuthority value = -100 }
	}
}

# Yearly income. Run in the king's scope.
# Crown baseline 10 (+10 at Royal Authority 30, +10 at 60) comes from the crown lands themselves.
# Skim: a share of real Earth Kingdom tribute moved from the king's purse into the treasury:
# outpost 1, territory 2, province 3, state 4 gold per tributary (+25% after A Single Law for the Kingdom),
# never more than 25% of the king's gold, never more than 80 a year together with the baseline,
# and nothing at all while the king has less than 25 gold.
atxb_yearly_income_effect = {
	set_variable = { which = global_atxb_last_crown value = 10 }
	if = {
		limit = { check_variable = { which = global_royalAuthority value = 30 } }
		change_variable = { which = global_atxb_last_crown value = 10 }
	}
	if = {
		limit = { check_variable = { which = global_royalAuthority value = 60 } }
		change_variable = { which = global_atxb_last_crown value = 10 }
	}
	change_variable = { which = global_atxb_treasury which = global_atxb_last_crown }

	set_variable = { which = global_atxb_last_skim value = 0 }
	set_variable = { which = global_atxb_tribs value = 0 }
	any_tributary = {
		limit = { is_tributary = { type = earth_kingdom_outpost_tributary } }
		change_variable = { which = global_atxb_last_skim value = 1 }
		change_variable = { which = global_atxb_tribs value = 1 }
	}
	any_tributary = {
		limit = { is_tributary = { type = earth_kingdom_territory_tributary } }
		change_variable = { which = global_atxb_last_skim value = 2 }
		change_variable = { which = global_atxb_tribs value = 1 }
	}
	any_tributary = {
		limit = { is_tributary = { type = earth_kingdom_province_tributary } }
		change_variable = { which = global_atxb_last_skim value = 3 }
		change_variable = { which = global_atxb_tribs value = 1 }
	}
	any_tributary = {
		limit = { is_tributary = { type = earth_kingdom_state_tributary } }
		change_variable = { which = global_atxb_last_skim value = 4 }
		change_variable = { which = global_atxb_tribs value = 1 }
	}
	if = {
		limit = { has_global_flag = atxb_done_law }
		multiply_variable = { which = global_atxb_last_skim value = 1.25 }
	}
	# cap: 80 a year including the crown baseline
	set_variable = { which = global_atxb_cap value = 80 }
	subtract_variable = { which = global_atxb_cap which = global_atxb_last_crown }
	if = {
		limit = { check_variable = { which = global_atxb_last_skim which = global_atxb_cap } }
		set_variable = { which = global_atxb_last_skim which = global_atxb_cap }
	}
	# cap: 25% of the king's gold, and nothing below 25 gold
	export_to_variable = { which = global_atxb_purse value = wealth }
	multiply_variable = { which = global_atxb_purse value = 0.25 }
	if = {
		limit = { check_variable = { which = global_atxb_last_skim which = global_atxb_purse } }
		set_variable = { which = global_atxb_last_skim which = global_atxb_purse }
	}
	if = {
		limit = { NOT = { wealth = 25 } }
		set_variable = { which = global_atxb_last_skim value = 0 }
	}
	# move the gold, whole coins only
	set_variable = { which = global_atxb_left which = global_atxb_last_skim }
	while = {
		limit = { check_variable = { which = global_atxb_left value = 10 } }
		wealth = -10
		change_variable = { which = global_atxb_treasury value = 10 }
		change_variable = { which = global_atxb_left value = -10 }
	}
	while = {
		limit = { check_variable = { which = global_atxb_left value = 1 } }
		wealth = -1
		change_variable = { which = global_atxb_treasury value = 1 }
		change_variable = { which = global_atxb_left value = -1 }
	}
	subtract_variable = { which = global_atxb_last_skim which = global_atxb_left }
	atxb_clamp_treasury_effect = yes
}

}

# A puppet Earth King does not spend the treasury: the Grand Secretariat does, in its own interest.
# It first saves for the Royal Granaries and builds them once (they keep the Lower Ring quiet), never builds the roads or
# the single law, and once the Granaries stand it takes a quarter of the treasury every year. The Secretariat is the holder of d_dai_li where that title
# exists, otherwise the king's chancellor. Run in the king's scope, after income and work stages.
atxb_secretariat_effect = {
	atxb_find_secretariat_effect = yes
	# Saves until the treasury covers the whole cost (100 + 2 x 100), so no stage ever stalls
	if = {
		limit = {
			atxb_any_work_running_trigger = no
			NOT = { has_global_flag = atxb_done_granaries }
			check_variable = { which = global_atxb_treasury value = 300 }
		}
		atxb_start_granaries_effect = yes
		set_character_flag = atxb_sec_started
	}
	# Skims only once the Granaries are built
	if = {
		limit = {
			has_global_flag = atxb_done_granaries
			check_variable = { which = global_atxb_treasury value = 50 }
		}
		set_variable = { which = global_atxb_sec_take which = global_atxb_treasury }
		multiply_variable = { which = global_atxb_sec_take value = 0.25 }
		if = {
			limit = { NOT = { check_variable = { which = global_atxb_sec_take value = 25 } } }
			set_variable = { which = global_atxb_sec_take value = 25 }
		}
		set_variable = { which = global_atxb_left which = global_atxb_sec_take }
		while = {
			limit = { check_variable = { which = global_atxb_left value = 10 } }
			change_variable = { which = global_atxb_treasury value = -10 }
			change_variable = { which = global_atxb_left value = -10 }
			event_target:atxb_secretariat = { wealth = 10 }
		}
		while = {
			limit = { check_variable = { which = global_atxb_left value = 1 } }
			change_variable = { which = global_atxb_treasury value = -1 }
			change_variable = { which = global_atxb_left value = -1 }
			event_target:atxb_secretariat = { wealth = 1 }
		}
		subtract_variable = { which = global_atxb_sec_take which = global_atxb_left }
		set_character_flag = atxb_sec_drew
		log = "ATLA Ba Sing Se: the Grand Secretariat took [global_atxb_sec_take.GetValue] from the treasury"
	}
	if = {
		limit = {
			ai = no
			OR = {
				has_character_flag = atxb_sec_started
				has_character_flag = atxb_sec_drew
			}
		}
		character_event = { id = atxb.6 }
	}
	else = {
		clr_character_flag = atxb_sec_started
		clr_character_flag = atxb_sec_drew
	}
}

# Saves event_target:atxb_secretariat: the holder of d_dai_li, otherwise the king's chancellor (king's scope)
atxb_find_secretariat_effect = {
	if = {
		limit = { d_dai_li = { has_holder = yes } }
		d_dai_li = { holder_scope = { save_event_target_as = atxb_secretariat } }
	}
	else = {
		job_chancellor = { save_event_target_as = atxb_secretariat }
	}
}

# Work completions (king's scope)
atxb_complete_roads_effect = {
	set_global_flag = atxb_done_roads
	any_province = {
		limit = {
			has_owner = yes
			region = world_greater_earth_kingdom
			NOT = { region = world_si_wong }
		}
		add_province_modifier = { name = atxb_royal_road duration = -1 }
	}
	if = {
		limit = { ai = no }
		set_character_flag = atxb_done_roads
		character_event = { id = atxb.15 days = 0 }
	}
}
atxb_complete_law_effect = {
	set_global_flag = atxb_done_law
	change_variable = { which = global_royalAuthority value = 5 }
	atxb_clamp_authority_effect = yes
	if = {
		limit = { ai = no }
		set_character_flag = atxb_done_law
		character_event = { id = atxb.15 days = 1 }
	}
	any_tributary = {
		limit = { atxb_ek_tributary_trigger = yes }
		opinion = { who = PREV modifier = opinion_atxb_single_law years = 20 }
		if = {
			limit = { ai = no }
			character_event = { id = atxb.16 days = 3 }
		}
	}
}

	set_global_flag = atxb_done_granaries

		limit = { region = world_ba_sing_se }
		add_province_modifier = { name = atxb_granaries duration = -1 }
	}
	if = {
		limit = { ai = no }
		set_character_flag = atxb_done_granaries
		character_event = { id = atxb.15 days = 2 }
	}
}

# Looting: everything in the treasury goes to the looter (character scope)
atxb_loot_treasury_effect = {
	set_variable = { which = global_atxb_last_loot which = global_atxb_treasury }
	while = {
		limit = { check_variable = { which = global_atxb_treasury value = 10 } }
		wealth = 10
		change_variable = { which = global_atxb_treasury value = -10 }
	}
	while = {
		limit = { check_variable = { which = global_atxb_treasury value = 1 } }
		wealth = 1
		change_variable = { which = global_atxb_treasury value = -1 }
	}
	set_variable = { which = global_atxb_treasury value = 0 }
}

# Sabotage sets a running work back one stage (first running work: roads, law, granaries)
atxb_sabotage_setback_effect = {
	if = {
		limit = { check_variable = { which = global_atxb_w_roads_stage value = 2 } }
		change_variable = { which = global_atxb_w_roads_stage value = -1 }
		set_character_flag = atxb_sab_roads
	}
	else_if = {
		limit = { check_variable = { which = global_atxb_w_law_stage value = 2 } }
		change_variable = { which = global_atxb_w_law_stage value = -1 }
		set_character_flag = atxb_sab_law
	}
	else_if = {
		limit = { check_variable = { which = global_atxb_w_granaries_stage value = 2 } }
		change_variable = { which = global_atxb_w_granaries_stage value = -1 }
		set_character_flag = atxb_sab_granaries
	}
	else = {
		# a work still in its first stage loses a year to a stall instead
		if = {
			limit = { check_variable = { which = global_atxb_w_roads_stage value = 1 } }
			change_variable = { which = global_atxb_w_roads_stall value = 1 }
			set_character_flag = atxb_sab_roads
		}
		else_if = {
			limit = { check_variable = { which = global_atxb_w_law_stage value = 1 } }
			change_variable = { which = global_atxb_w_law_stall value = 1 }
			set_character_flag = atxb_sab_law
		}
		else_if = {
			limit = { check_variable = { which = global_atxb_w_granaries_stage value = 1 } }
			change_variable = { which = global_atxb_w_granaries_stall value = 1 }
			set_character_flag = atxb_sab_granaries
		}
	}
}

# A tributary breaks away: exactly the effect of the base mod's break_away_from_earth_kingdom decision.
# Run in an event whose ROOT is the tributary (atxb.20), so FROM in the base events is the right character.
atxb_breakaway_effect = {
	log = "ATLA Ba Sing Se: [Root.GetTitledName] broke away from the Earth Kingdom"
	suzerain = {
		remove_tributary = PREV
		character_event = { id = ATLAE.208 }
	}
	character_event = { id = ATLAE.209 }
	set_character_flag = earth_kingdom_rebel
	any_independent_ruler = {
		limit = {
			culture_group = earth_kingdom_cultures
			is_tributary = yes
		}
		character_event = { id = ATLAE.210 }
	}
}

# ---- v2: the reach of Ba Sing Se ----
# Marks every Earth tributary connected to Ba Sing Se by a chain of tributaries (flag atxb_linked).
# Land: spread out from the king's realm through bordering tributaries (at most 20 rings).
# Sea: island tributaries (world_earth_islands) count as connected while Ba Sing Se or a connected tributary holds a port.
# Run in the king's scope. atxb_unlink_all_effect clears the flags again at the end of the yearly run.
atxb_flood_fill_effect = {
	any_tributary = { clr_character_flag = atxb_linked }
	set_character_flag = atxb_linked
	set_variable = { which = global_atxb_rings value = 0 }
	atxb_flood_land_effect = yes
	if = {
		limit = {
			OR = {
				any_realm_province = { port = yes }
				any_tributary = {
					has_character_flag = atxb_linked
					any_realm_province = { port = yes }
				}
			}
		}
		any_tributary = {
			limit = {
				atxb_ek_tributary_trigger = yes
				NOT = { has_character_flag = atxb_linked }
				capital_scope = { region = world_earth_islands }
			}
			set_character_flag = atxb_linked
		}
		atxb_flood_land_effect = yes
	}
	set_variable = { which = global_atxb_unlinked value = 0 }
	any_tributary = {
		limit = {
			atxb_ek_tributary_trigger = yes
			NOT = { has_character_flag = atxb_linked }
		}
		change_variable = { which = global_atxb_unlinked value = 1 }
	}
}
atxb_flood_land_effect = {
	while = {
		limit = {
			NOT = { check_variable = { which = global_atxb_rings value = 20 } }
			any_tributary = {
				atxb_ek_tributary_trigger = yes
				NOT = { has_character_flag = atxb_linked }
				any_neighbor_independent_ruler = { has_character_flag = atxb_linked }
			}
		}
		change_variable = { which = global_atxb_rings value = 1 }
		any_tributary = {
			limit = {
				atxb_ek_tributary_trigger = yes
				NOT = { has_character_flag = atxb_linked }
				any_neighbor_independent_ruler = { has_character_flag = atxb_linked }
			}
			set_character_flag = atxb_linked
		}
	}
}
atxb_unlink_all_effect = {
	any_tributary = { clr_character_flag = atxb_linked }
	clr_character_flag = atxb_linked
}

# Notices a change of the Earth Kingdom unity law (global flag atxb_law_changed, for the 5-year grace)
atxb_law_watch_effect = {
	set_variable = { which = global_atxb_unity_now value = 0 }
	if = {
		limit = { k_earth_kingdom = { has_law = ek_unity_1 } }
		set_variable = { which = global_atxb_unity_now value = 1 }
	}
	if = {
		limit = { k_earth_kingdom = { has_law = ek_unity_2 } }
		set_variable = { which = global_atxb_unity_now value = 2 }
	}
	if = {
		limit = { k_earth_kingdom = { has_law = ek_unity_3 } }
		set_variable = { which = global_atxb_unity_now value = 3 }
	}
	if = {
		limit = {
			has_global_flag = atxb_unity_seen
			NOT = {
				AND = {
					check_variable = { which = global_atxb_unity_now which = global_atxb_unity_last }
					check_variable = { which = global_atxb_unity_last which = global_atxb_unity_now }
				}
			}
		}
		clr_global_flag = atxb_law_changed
		set_global_flag = atxb_law_changed
		log = "ATLA Ba Sing Se: the unity law changed, 5 years' grace for tributaries"
	}
	set_variable = { which = global_atxb_unity_last which = global_atxb_unity_now }
	if = {
		limit = { NOT = { has_global_flag = atxb_unity_seen } }
		set_global_flag = atxb_unity_seen
	}
}

# ---- v2: Plea to Ba Sing Se ----
# The pleader (character scope) sends its plea to the ruler of Ba Sing Se.
# A puppet king never sees it: the Grand Secretariat answers instead (atxb_secretariat_answer_plea_effect).
atxb_send_plea_effect = {
	atxb_find_king_effect = yes
	if = {
		limit = {
			event_target:atxb_king = { always = yes }
			NOT = { war_with = event_target:atxb_king }
		}
		clr_character_flag = atxb_pleaded
		set_character_flag = atxb_pleaded
		log = "ATLA Ba Sing Se: [Root.GetTitledName] pleaded with Ba Sing Se for help"
		if = {
			limit = { event_target:atxb_king = { trait = earthkingpuppet } }
			atxb_secretariat_answer_plea_effect = yes
		}
		else = {
			event_target:atxb_king = { character_event = { id = atxb.31 } }
		}
	}
}

# The Grand Secretariat answers a plea for a puppet king: three times in four it turns the envoy away,
# otherwise it sends a token host of 300 (30 from the treasury, if the treasury has it). Pleader's scope.
atxb_secretariat_answer_plea_effect = {
	random_list = {
		75 = {
			set_character_flag = atxb_ans_sec_refused
			atxb_plea_refused_effect = yes
		}
		25 = {
			modifier = { factor = 0 NOT = { check_variable = { which = global_atxb_treasury value = 30 } } }
			set_character_flag = atxb_ans_sec_token
			change_variable = { which = global_atxb_treasury value = -30 }
			atxb_raise_host_small_effect = yes
			opinion = { who = event_target:atxb_king modifier = opinion_atxb_token_help years = 5 }
			clr_character_flag = atxb_helped_plea
			set_character_flag = atxb_helped_plea
		}
	}
	event_target:atxb_king = {
		character_event = { id = atxb.34 } # sends atxb.32 to the pleader, so the letter comes from the king
		if = {
			limit = { ai = no }
			character_event = { id = atxb.33 days = 8 }
		}
	}
}

# ---- Overextended Crown (character scope: the Earth King) ----
# Counts the king's realm provinces outside the Ba Sing Se region and sets the matching modifier.
# Capacity: 15 / 45, +15 to both for the Royal Roads, +15 to both for A Single Law.
# Monarchism (ek_unity_3) with both works finished lifts it completely.
# The modifier lasts 400 days and is renewed every year, so a former king loses it on his own.
atxb_overextension_effect = {
	set_variable = { which = global_atxb_outside value = 0 }
	any_realm_province = {
		limit = { NOT = { region = world_ba_sing_se } }
		change_variable = { which = global_atxb_outside value = 1 }
	}
	set_variable = { which = global_atxb_cap1 value = 15 }
	set_variable = { which = global_atxb_cap2 value = 45 }
	if = {
		limit = { has_global_flag = atxb_done_roads }
		change_variable = { which = global_atxb_cap1 value = 15 }
		change_variable = { which = global_atxb_cap2 value = 15 }
	}
	if = {
		limit = { has_global_flag = atxb_done_law }
		change_variable = { which = global_atxb_cap1 value = 15 }
		change_variable = { which = global_atxb_cap2 value = 15 }
	}
	remove_character_modifier = atxb_overextended_1
	remove_character_modifier = atxb_overextended_2
	clr_global_flag = atxb_overext_lifted
	clr_global_flag = atxb_overext_1
	clr_global_flag = atxb_overext_2
	if = {
		limit = {
			primary_title = { has_law = ek_unity_3 }
			has_global_flag = atxb_done_roads
			has_global_flag = atxb_done_law
		}
		set_global_flag = atxb_overext_lifted
	}
	else_if = {
		limit = { check_variable = { which = global_atxb_outside which = global_atxb_cap2 } }
		add_character_modifier = { name = atxb_overextended_2 duration = 400 }
		set_global_flag = atxb_overext_2
	}
	else_if = {
		limit = { check_variable = { which = global_atxb_outside which = global_atxb_cap1 } }
		add_character_modifier = { name = atxb_overextended_1 duration = 400 }
		set_global_flag = atxb_overext_1
	}
}

# Overextended Crown readout (custom tooltips; values from the last yearly run). Used by atxb.5 and ATLA Dev Test.
atxb_overext_tooltips = {
	if = {
		limit = { has_global_flag = atxb_bss_started }
		if = {
			limit = { has_global_flag = atxb_overext_lifted }
			custom_tooltip = { text = atxb_overext_lifted_tt }
		}
		else = {
			custom_tooltip = { text = atxb_overext_count_tt }
			if = {
				limit = { has_global_flag = atxb_overext_2 }
				custom_tooltip = { text = atxb_overext_2_tt }
			}
			else_if = {
				limit = { has_global_flag = atxb_overext_1 }
				custom_tooltip = { text = atxb_overext_1_tt }
			}
		}
	}
}

# ---- v3: Royal Arbitration (ROOT = the attacker, event_target:atxb_arbiter = the Earth King) ----
# Ends every war in which ROOT attacks a non-Chin Earth realm (never a war against the Fire Nation), and ROOT may not declare war for 2 years
atxb_arb_end_wars_effect = {
	any_war = {
		limit = {
			attacker = { character = ROOT }
			defender = {
				culture_group = earth_kingdom_cultures
				NOT = { religion = way_of_chin }
			}
		}
		end_war = whitepeace
	}
	set_pacified = { years = 2 }
}
