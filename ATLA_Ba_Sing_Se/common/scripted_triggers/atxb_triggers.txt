# ATLA Ba Sing Se - triggers (prefix atxb_)
# Builds on the base mod's own Earth Kingdom tributaries (common/tributary_types/ATLA_tributary_types.txt),
# Royal Authority (global_royalAuthority) and puppet Earth King (trait earthkingpuppet).

# An Earth Kingdom ruler who may hold Ba Sing Se's crown (not a Way of Chin or foreign conqueror)
atxb_valid_king_trigger = {
	independent = yes
	culture_group = earth_kingdom_cultures
	religion_group = earth_religions
	NOT = { religion = way_of_chin }
}

# This character is the ruler of Ba Sing Se (holder of e_earth_kingdom, or of k_earth_kingdom while the empire is vacant)
atxb_is_king_trigger = {
	atxb_valid_king_trigger = yes
	OR = {
		has_landed_title = e_earth_kingdom
		AND = {
			has_landed_title = k_earth_kingdom
			NOT = { e_earth_kingdom = { has_holder = yes } }
		}
	}
}

# The ruler of Ba Sing Se may spend the treasury himself (a puppet king's treasury is spent by the Grand Secretariat)
atxb_king_can_spend_trigger = {
	atxb_is_king_trigger = yes
	NOT = { trait = earthkingpuppet }
}

# One of the base mod's four Earth Kingdom tributary types (character scope, the tributary)
atxb_ek_tributary_trigger = {
	OR = {
		is_tributary = { type = earth_kingdom_outpost_tributary }
		is_tributary = { type = earth_kingdom_territory_tributary }
		is_tributary = { type = earth_kingdom_province_tributary }
		is_tributary = { type = earth_kingdom_state_tributary }
	}
}

# A work is under construction
atxb_any_work_running_trigger = {
	OR = {
		check_variable = { which = global_atxb_w_roads_stage value = 1 }
		check_variable = { which = global_atxb_w_law_stage value = 1 }
		check_variable = { which = global_atxb_w_granaries_stage value = 1 }
	}
}

# The crown is weak enough for tributaries to break away (needs event_target:atxb_king)
atxb_crown_weak_trigger = {
	OR = {
		event_target:atxb_king = { trait = earthkingpuppet }
		NOT = { check_variable = { which = global_royalAuthority value = 30 } }
		AND = {
			event_target:atxb_king = { war = yes }
			NOT = { check_variable = { which = global_royalAuthority value = 50 } }
		}
	}
}

# An AI tributary that may break away (character scope, the tributary; needs event_target:atxb_king)
# An AI tributary that may break away (character scope, the tributary; needs event_target:atxb_king).
# Close to Ba Sing Se it takes a weak crown and a tributary that dislikes the king.
# Out of reach (no chain of tributaries to Ba Sing Se, flag atxb_linked missing) it only takes opinion below 25.
atxb_breakaway_candidate_trigger = {
	ai = yes
	higher_tier_than = COUNT
	prisoner = no
	atxb_ek_tributary_trigger = yes
	# grace periods: 3 years after the campaign starts, 5 after a unity-law change, 10 after a plea the crown answered
	had_global_flag = { flag = atxb_bss_started days = 1095 }
	OR = {
		NOT = { has_global_flag = atxb_law_changed }
		had_global_flag = { flag = atxb_law_changed days = 1825 }
	}
	OR = {
		NOT = { has_character_flag = atxb_helped_plea }
		had_character_flag = { flag = atxb_helped_plea days = 3650 }
	}
	OR = {
		AND = {
			NOT = { opinion = { who = event_target:atxb_king value = 0 } }
			atxb_crown_weak_trigger = yes
		}
		AND = {
			NOT = { has_character_flag = atxb_linked }
			NOT = { opinion = { who = event_target:atxb_king value = 25 } }
		}
	}
}

# An Earth realm that may plead with Ba Sing Se (character scope). Works without event targets, so decisions can use it.
atxb_plea_base_trigger = {
	is_ruler = yes
	independent = yes
	prisoner = no
	higher_tier_than = COUNT
	culture_group = earth_kingdom_cultures
	religion_group = earth_religions
	NOR = {
		religion = way_of_chin
		culture = sandbender_culture
		has_landed_title = e_earth_kingdom
		has_landed_title = k_earth_kingdom
		mercenary = yes
		holy_order = yes
		rebel = yes
	}
	capital_scope = {
		region = world_greater_earth_kingdom
		NOT = { region = world_si_wong }
	}
	OR = {
		e_earth_kingdom = { has_holder = yes holder_scope = { atxb_is_king_trigger = yes } }
		k_earth_kingdom = { has_holder = yes holder_scope = { atxb_is_king_trigger = yes } }
	}
}

# ROOT is defending against a foreign (Fire Nation), Way of Chin or rebel attacker, is not at war with the Earth King,
# and has not pleaded in the last 5 years
atxb_plea_war_trigger = {
	NOR = {
		e_earth_kingdom = { has_holder = yes holder_scope = { war_with = ROOT } }
		k_earth_kingdom = { has_holder = yes holder_scope = { war_with = ROOT } }
	}
	any_war = {
		defender = { character = ROOT }
		attacker = {
			OR = {
				culture_group = fire_nation_cultures
				religion = way_of_chin
				rebel = yes
			}
		}
	}
	OR = {
		NOT = { has_character_flag = atxb_pleaded }
		had_character_flag = { flag = atxb_pleaded days = 1825 }
	}
}

# ROOT dislikes the crown or is the king's rival (gate for AI sabotage)
atxb_hostile_to_crown_trigger = {
	OR = {
		religion = way_of_chin
		e_earth_kingdom = {
			has_holder = yes
			holder_scope = {
				atxb_is_king_trigger = yes
				OR = {
					is_rival = ROOT
					ROOT = { NOT = { opinion = { who = PREV value = 0 } } }
				}
			}
		}
		k_earth_kingdom = {
			has_holder = yes
			holder_scope = {
				atxb_is_king_trigger = yes
				OR = {
					is_rival = ROOT
					ROOT = { NOT = { opinion = { who = PREV value = 0 } } }
				}
			}
		}
	}
}

# ---- v3: the Treasury of Ba Sing Se (one popup for players, decisions for the AI) ----
# The king may start a work now (also checks the cost)
atxb_can_start_roads_trigger = {
	atxb_king_can_spend_trigger = yes
	NOT = { has_global_flag = atxb_done_roads }
	NOT = { check_variable = { which = global_atxb_w_roads_stage value = 1 } }
	check_variable = { which = global_atxb_treasury value = 100 }
}
atxb_can_start_law_trigger = {
	atxb_king_can_spend_trigger = yes
	NOT = { has_global_flag = atxb_done_law }
	NOT = { check_variable = { which = global_atxb_w_law_stage value = 1 } }
	check_variable = { which = global_atxb_treasury value = 200 }
	atxb_law_unity_ok_trigger = yes
}
# A Single Law needs the States or Monarchism unity law (on the empire or the kingdom)
atxb_law_unity_ok_trigger = {
	OR = {
		e_earth_kingdom = { OR = { has_law = ek_unity_2 has_law = ek_unity_3 } }
		k_earth_kingdom = { OR = { has_law = ek_unity_2 has_law = ek_unity_3 } }
	}
}
# A work is neither finished nor under construction (the works page is worth showing)
atxb_any_work_open_trigger = {
	atxb_king_can_spend_trigger = yes
	OR = {
		AND = {
			NOT = { has_global_flag = atxb_done_roads }
			NOT = { check_variable = { which = global_atxb_w_roads_stage value = 1 } }
		}
		AND = {
			NOT = { has_global_flag = atxb_done_law }
			NOT = { check_variable = { which = global_atxb_w_law_stage value = 1 } }
		}
		AND = {
			NOT = { has_global_flag = atxb_done_granaries }
			NOT = { check_variable = { which = global_atxb_w_granaries_stage value = 1 } }
		}
	}
}
atxb_can_start_granaries_trigger = {
	atxb_king_can_spend_trigger = yes
	NOT = { has_global_flag = atxb_done_granaries }
	NOT = { check_variable = { which = global_atxb_w_granaries_stage value = 1 } }
	check_variable = { which = global_atxb_treasury value = 100 }
}
# Draw on the Treasury: 100 out, every 2 years
atxb_can_draw_trigger = {
	atxb_king_can_spend_trigger = yes
	check_variable = { which = global_atxb_treasury value = 100 }
	OR = {
		NOT = { has_character_flag = atxb_drew }
		had_character_flag = { flag = atxb_drew days = 730 }
	}
}
# Corvee: Royal Authority 15 for treasury +100, while a work is running, once a year
atxb_can_corvee_trigger = {
	atxb_king_can_spend_trigger = yes
	atxb_any_work_running_trigger = yes
	check_variable = { which = global_royalAuthority value = 15 }
	OR = {
		NOT = { has_character_flag = atxb_corvee }
		had_character_flag = { flag = atxb_corvee days = 365 }
	}
}

# ---- v3: Royal Arbitration ----
# ROOT is an Earth realm attacking another Earth realm, and FROM (the Earth King) is in neither side of it
atxb_arbitration_target_trigger = {
	is_ruler = yes
	independent = yes
	prisoner = no
	higher_tier_than = COUNT
	culture_group = earth_kingdom_cultures
	NOR = {
		character = FROM
		religion = way_of_chin
		mercenary = yes
		holy_order = yes
		rebel = yes
		war_with = FROM
	}
	capital_scope = { region = world_greater_earth_kingdom }
	any_war = {
		attacker = { character = ROOT }
		defender = {
			culture_group = earth_kingdom_cultures
			rebel = no
			NOT = { religion = way_of_chin }
			NOT = { character = FROM }
		}
	}
}
