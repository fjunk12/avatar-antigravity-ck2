character_event = {
	id = atxd.2
	desc = atxd.2.d
	picture = GFX_evt_ba_sing_se
	is_triggered_only = yes

	immediate = {
		atxb_flood_fill_effect = yes
		log = "ATLA Dev Test: reach check for [Root.GetTitledName]: [global_atxb_unlinked.GetValue] tributaries out of reach"
		any_tributary = {
			limit = { atxb_ek_tributary_trigger = yes has_character_flag = atxb_linked }
			log = "ATLA Dev Test: connected [This.GetTitledName]"
		}
		any_tributary = {
			limit = { atxb_ek_tributary_trigger = yes NOT = { has_character_flag = atxb_linked } }
			log = "ATLA Dev Test: OUT OF REACH [This.GetTitledName]"
		}
		set_variable = { which = global_atxd_linked value = 0 }
		any_tributary = {
			limit = { atxb_ek_tributary_trigger = yes has_character_flag = atxb_linked }
			change_variable = { which = global_atxd_linked value = 1 }
		}
	}

	option = {
		name = atxd.2.a
		atxb_unlink_all_effect = yes
	}
}
