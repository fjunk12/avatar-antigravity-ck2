# ATLA Fire Nation - events (namespace atxf)
# atxf.1   yearly run, once a year for the whole world (on_yearly_pulse, hidden)
# atxf.2   the Camellia-Peony dispute begins (to both claimants)
# atxf.3   a Fire Isles lord takes a side
# atxf.4   the Avatar's warning (letter to both claimants)
# atxf.5   hidden relay on the Avatar: sends atxf.4 to both claimants, so the letter comes from the Avatar
# atxf.10  an envoy courts a lord for the other side (capped)
# atxf.30  the aftermath: the fate of the clans (to the winner)
# atxf.50  the project to break the clans: a hard step (to the Fire Lord)
# atxf.52  a clan remonstrance (backlash while the project runs)
# atxf.53  the Absolute Throne: the project is complete
# atxf.54  the Avatar's demand for clemency (letter to the Fire Lord, FROM = the Avatar)
# atxf.55  the clan council's demand under the Compact
# atxf.56  hidden relay on the Avatar: sends atxf.54
# atxf.60  the Fire Sages and the throne (once per Fire Lord)
# atxf.40  Petition the Fire Lord: a player chooses what to ask (decision)
# atxf.41  Petition the Fire Lord: the Fire Lord answers (ROOT = Fire Lord, FROM = petitioner)
# atxf.42  Petition the Fire Lord: the answer letter (ROOT = petitioner, FROM = Fire Lord)
# atxf.43  Petition the Fire Lord: an AI vassal petitions (hidden, at most 2 a year)
namespace = atxf

character_event = {
	id = atxf.1
	hide_window = yes
	is_triggered_only = yes # on_yearly_pulse

	only_rulers = yes

	trigger = {
		OR = {
			NOT = { has_global_flag = atxf_tick }
			had_global_flag = { flag = atxf_tick days = 364 } # once a year: rulers pulse every day, so this fires on day 365
		}
	}

	immediate = {
		clr_global_flag = atxf_tick
		set_global_flag = atxf_tick
		set_variable = { which = global_atxf_petitions value = 0 }
		set_variable = { which = global_atxf_courts value = 0 }

		# ---- The Camellia-Peony War ----
		if = {
			limit = { atxf_cpw_can_start_trigger = yes }
			atxf_cpw_start_effect = yes
		}
		else_if = {
			limit = { has_global_flag = atxf_cpw }
			atxf_find_champions_effect = yes
			if = {
				limit = { atxf_cpw_decided_trigger = yes }
				log = "ATLA Fire Nation: the Camellia-Peony dispute is decided"
				atxf_cpw_cleanup_effect = yes
				if = {
					limit = { e_fire_nation = { has_holder = yes } }
					e_fire_nation = { holder_scope = { character_event = { id = atxf.30 } } }
				}
				else_if = {
					limit = { k_fire_nation = { has_holder = yes } }
					k_fire_nation = { holder_scope = { character_event = { id = atxf.30 } } }
				}
				else = {
					k_western_fire_nation = { holder_scope = { character_event = { id = atxf.30 } } }
				}
			}
			else_if = {
				limit = { year = 600 }
				log = "ATLA Fire Nation: the Camellia-Peony dispute fades without a victor"
				atxf_cpw_cleanup_effect = yes
			}
			else = {
				atxf_call_of_the_clans_effect = yes
				# Court intrigue: at most 2 envoys a year, at most one per lord every 2 years
				random_character = {
					limit = {
						atxf_cpw_lord_trigger = yes
						NOT = { dynasty = 3994 }
						NOT = { dynasty = 3995 }
						OR = {
							NOT = { has_character_flag = atxf_courted }
							had_character_flag = { flag = atxf_courted days = 730 }
						}
					}
					character_event = { id = atxf.10 }
				}
				random = {
					chance = 50
					random_character = {
						limit = {
							atxf_cpw_lord_trigger = yes
							NOT = { dynasty = 3994 }
							NOT = { dynasty = 3995 }
							OR = {
								NOT = { has_character_flag = atxf_courted }
								had_character_flag = { flag = atxf_courted days = 730 }
							}
						}
						character_event = { id = atxf.10 }
					}
				}
			}
		}

		# ---- The Fire Lord: the Fire Sages, the clan project, the Compact ----
		atxf_find_fire_lord_effect = yes
		if = {
			limit = {
				event_target:atxf_fire_lord = { culture_group = fire_nation_cultures }
			}
			# The Fire Sages ask every new Fire Lord where they stand
			event_target:atxf_fire_lord = {
				if = {
					limit = { NOT = { has_character_flag = atxf_sages_asked } }
					set_character_flag = atxf_sages_asked
					character_event = { id = atxf.60 days = 30 }
				}
			}
			# Breaking the clans: a generational project (+4 a year, more with every hard step)
			if = {
				limit = { atxf_project_running_trigger = yes }
				change_variable = { which = global_atxf_centralization value = 4 }
				atxf_clamp_project_effect = yes
				if = {
					limit = { check_variable = { which = global_atxf_centralization value = 100 } }
					set_global_flag = atxf_throne_absolute
					log = "ATLA Fire Nation: the clans are broken; the Absolute Throne is established"
					event_target:atxf_fire_lord = { character_event = { id = atxf.53 } }
				}
				else = {
					log = "ATLA Fire Nation: the project to break the clans stands at [global_atxf_centralization.GetValue]/100"
					random = {
						chance = 33
						event_target:atxf_fire_lord = { character_event = { id = atxf.50 days = 30 random = 200 } }
					}
					# Backlash: an aggrieved clan lord remonstrates
					random = {
						chance = 20
						random_character = {
							limit = {
								atxf_clan_lord_trigger = yes
								NOT = { opinion = { who = event_target:atxf_fire_lord value = 0 } }
							}
							save_global_event_target_as = atxf_remonstrant
							event_target:atxf_fire_lord = { character_event = { id = atxf.52 days = 30 random = 200 } }
						}
					}
				}
			}
			# The Absolute Throne passes to every Fire Lord
			if = {
				limit = { has_global_flag = atxf_throne_absolute }
				event_target:atxf_fire_lord = {
					remove_character_modifier = atxf_absolute_throne
					add_character_modifier = { name = atxf_absolute_throne duration = 400 }
				}
			}
			# The Compact passes to every Fire Lord while it holds, and the clan council makes demands
			if = {
				limit = { has_global_flag = atxf_path_compact }
				event_target:atxf_fire_lord = {
					remove_character_modifier = atxf_clan_compact
					add_character_modifier = { name = atxf_clan_compact duration = 400 }
				}
				random = {
					chance = 33
					event_target:atxf_fire_lord = { character_event = { id = atxf.55 days = 30 random = 200 } }
				}
			}
		}

		# ---- Petition the Fire Lord: AI vassals (at most 2 a year, counted in atxf.43) ----
		e_fire_nation = {
			holder_scope = {
				any_vassal = {
					limit = { ai = yes atxf_petition_base_trigger = yes }
					random = { chance = 5 character_event = { id = atxf.43 days = 10 random = 60 } }
				}
			}
		}
		if = {
			limit = { NOT = { e_fire_nation = { has_holder = yes } } }
			k_fire_nation = {
				holder_scope = {
					any_vassal = {
						limit = { ai = yes atxf_petition_base_trigger = yes }
						random = { chance = 5 character_event = { id = atxf.43 days = 10 random = 60 } }
					}
				}
			}
		}
	}
}

# An envoy of one side courts a lord (ROOT = the lord)
character_event = {
	id = atxf.10
	desc = {
		trigger = { has_character_flag = atxf_court_peony }
		text = atxf.10.d_peony
	}
	desc = {
		trigger = { has_character_flag = atxf_court_camellia }
		text = atxf.10.d_camellia
	}
	picture = GFX_evt_courtiers_talking
	is_triggered_only = yes

	trigger = {
		has_global_flag = atxf_cpw
		NOT = { check_variable = { which = global_atxf_courts value = 2 } }
	}

	immediate = {
		change_variable = { which = global_atxf_courts value = 1 }
		clr_character_flag = atxf_courted
		set_character_flag = atxf_courted
		atxf_find_champions_effect = yes
		# The other side does the courting; a neutral lord is courted by either
		if = {
			limit = { has_character_flag = atxf_peony }
			set_character_flag = atxf_court_camellia
		}
		else_if = {
			limit = { has_character_flag = atxf_camellia }
			set_character_flag = atxf_court_peony
		}
		else = {
			random_list = {
				50 = { set_character_flag = atxf_court_peony }
				50 = { set_character_flag = atxf_court_camellia }
			}
		}
	}

	option = {
		name = atxf.10.a # accept the gifts and change sides
		wealth = 25
		if = {
			limit = { has_character_flag = atxf_court_peony }
			atxf_join_peony_effect = yes
			event_target:atxf_peony_champion = {
				if = { limit = { wealth = 25 } wealth = -25 }
			}
		}
		else = {
			atxf_join_camellia_effect = yes
			event_target:atxf_camellia_champion = {
				if = { limit = { wealth = 25 } wealth = -25 }
			}
		}
		clr_character_flag = atxf_court_peony
		clr_character_flag = atxf_court_camellia
		log = "ATLA Fire Nation: [Root.GetTitledName] changed sides in the Camellia-Peony dispute"
		ai_chance = {
			factor = 30
			modifier = { factor = 2 has_character_flag = atxf_neutral }
			modifier = { factor = 1.5 trait = greedy }
			modifier = { factor = 0.5 trait = honest }
		}
	}
	option = {
		name = atxf.10.b # refuse
		if = {
			limit = { atxf_cpw_aligned_trigger = yes }
			prestige = 10
		}
		clr_character_flag = atxf_court_peony
		clr_character_flag = atxf_court_camellia
		ai_chance = { factor = 70 }
	}
}

# Petition the Fire Lord: an AI vassal petitions (ROOT = the petitioner). At most 2 a year.
character_event = {
	id = atxf.43
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { check_variable = { which = global_atxf_petitions value = 2 } }
	}

	immediate = {
		change_variable = { which = global_atxf_petitions value = 1 }
		clr_character_flag = atxf_pet_gold
		clr_character_flag = atxf_pet_denounce
		if = {
			limit = { atxf_has_denounceable_rival_trigger = yes }
			random_rival = {
				limit = {
					is_ruler = yes
					is_landed = yes
					top_liege = { atxf_is_fire_lord_trigger = yes }
					NOT = { atxf_is_fire_lord_trigger = yes }
				}
				save_event_target_as = atxf_denounced # chain-scoped: each petition keeps its own rival
			}
			set_character_flag = atxf_pet_denounce
		}
		else = {
			set_character_flag = atxf_pet_gold
		}
		atxf_send_petition_effect = yes
	}
}
